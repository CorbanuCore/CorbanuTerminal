+//! Corbanu: agent role files may replace a role's base instructions
+//! (`base_instructions`, the alias of `instructions`) instead of, or in
+//! addition to, `developer_instructions` (main commit baeb1bf86e).
+
+use codex_agent_roles::ResolvedAgentRoleFile;
+use codex_agent_roles::parse_agent_role_file_contents;
+use std::path::PathBuf;
+
+fn parse(contents: &str, role_name_hint: Option<&str>) -> std::io::Result<ResolvedAgentRoleFile> {
+    let base_dir = std::env::temp_dir();
+    let label: PathBuf = base_dir.join("agents").join("role.toml");
+    parse_agent_role_file_contents(contents, &label, &base_dir, role_name_hint)
+}
+
+fn error_message(result: std::io::Result<ResolvedAgentRoleFile>) -> String {
+    match result {
+        Ok(parsed) => panic!("expected an error, got {parsed:?}"),
+        Err(err) => err.to_string(),
+    }
+}
+
+#[test]
+fn discovered_role_file_accepts_base_instructions_without_developer_instructions() {
+    let parsed = parse(
+        "name = \"orc\"\ndescription = \"Worker\"\nbase_instructions = \"You are an orc.\"\n",
+        /*role_name_hint*/ None,
+    )
+    .expect("a role file with only base_instructions should parse");
+
+    assert_eq!(parsed.role_name, "orc");
+    assert_eq!(parsed.description.as_deref(), Some("Worker"));
+    assert_eq!(
+        parsed
+            .config
+            .get("base_instructions")
+            .and_then(|value| value.as_str()),
+        Some("You are an orc.")
+    );
+    assert_eq!(parsed.config.get("name"), None);
+    assert_eq!(parsed.config.get("description"), None);
+}
+
+#[test]
+fn blank_base_instructions_are_rejected() {
+    let message = error_message(parse(
+        "name = \"orc\"\ndeveloper_instructions = \"Do work.\"\nbase_instructions = \"   \"\n",
+        /*role_name_hint*/ None,
+    ));
+    assert!(
+        message.contains(".base_instructions cannot be blank"),
+        "unexpected error: {message}"
+    );
+
+    let message = error_message(parse(
+        "base_instructions = \"\"\n",
+        /*role_name_hint*/ Some("orc"),
+    ));
+    assert!(
+        message.contains(".base_instructions cannot be blank"),
+        "unexpected error: {message}"
+    );
+}
