+/// Signals that it was cancelled, then hands its next input poll to a detached poller and stays
+/// running until released, like a turn still inside guardian's graceful-stop wait. The poller is
+/// detached so the reproduction does not race the 100 ms hard-abort timer.
+struct StoppingTurnTask {
+    cancelled: std::sync::Mutex<Option<tokio::sync::oneshot::Sender<()>>>,
+    poll: std::sync::Mutex<Option<tokio::sync::oneshot::Receiver<()>>>,
+    polled: Polled,
+    polled_tx: std::sync::Mutex<Option<tokio::sync::oneshot::Sender<()>>>,
+    release: Arc<tokio::sync::Notify>,
+}
+
+impl SessionTask for StoppingTurnTask {
+    fn kind(&self) -> TaskKind {
+        TaskKind::Regular
+    }
+
+    fn span_name(&self) -> &'static str {
+        "session_task.stopping_turn"
+    }
+
+    async fn run(
+        self: Arc<Self>,
+        session: Arc<SessionTaskContext>,
+        ctx: Arc<TurnContext>,
+        _input: Vec<TurnInput>,
+        cancellation_token: CancellationToken,
+    ) -> SessionTaskResult {
+        cancellation_token.cancelled().await;
+        let session = session.clone_session();
+        let sub_id = ctx.sub_id.clone();
+        let poll = self.poll.lock().expect("poll lock").take().expect("poll");
+        let polled = Arc::clone(&self.polled);
+        let polled_tx = self
+            .polled_tx
+            .lock()
+            .expect("polled_tx lock")
+            .take()
+            .expect("polled_tx");
+        tokio::spawn(async move {
+            let _ = poll.await;
+            let (input, _) = session
+                .input_queue
+                .get_pending_input_for_running_task(&session.active_turn, &sub_id)
+                .await;
+            *polled.lock().expect("polled lock") = Some(input);
+            let _ = polled_tx.send(());
+        });
+        if let Some(cancelled) = self.cancelled.lock().expect("cancelled lock").take() {
+            let _ = cancelled.send(());
+        }
+        self.release.notified().await;
+        Ok(None)
+    }
+}

+## Agents and reviews
+
+- **A turn you start while guardian is stopping the previous one keeps its
+  input** (#2046). When automatic approval review stopped a turn after too many
+  denials, the old turn kept running briefly in the background. A turn you
+  started in that moment could lose the message you steered into it, and agent
+  messages meant for it, to the turn being stopped. The stopping turn now only
+  takes input that belongs to it.
