+    let polled = Polled::default();

+            poll: Arc::clone(&poll),

+    )
+    .await;
+
+    // Guardian's stop, detached as in `record_guardian_denial`.
+    let stop = tokio::spawn({

+            sess.abort_turn_if_active(&turn_id, TurnAbortReason::Interrupted)
+                .await
+        }
+    });
+    // Bounded: the stop takes the active slot before it waits for the task.
+    for _ in 0..1_000 {
+        if sess.active_turn.lock().await.is_none() {

+        }
+        tokio::task::yield_now().await;

+
+- **A turn started while guardian is stopping another keeps its own input

+  waits briefly for it to finish. A turn started in that window could have
+  its steering input and queued agent messages taken by the turn being
+  stopped. The stopping turn now takes only its own input.
