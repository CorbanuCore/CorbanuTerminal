+async fn guardian_stopping_turn_cannot_take_replacement_input() {
+    tokio::time::timeout(Duration::from_secs(10), async {
+        let (sess, old_turn, _rx) = make_session_and_context_with_rx().await;
+        let new_turn = sess
+            .new_default_turn_with_sub_id("replacement".to_string())
+            .await;
+        let polled = Arc::default();
+        let resume_poll = Arc::new(Notify::new());
+        sess.spawn_task(
+            Arc::clone(&old_turn),
+            Vec::new(),
+            StalePollTask {
+                polled: Arc::clone(&polled),
+                resume_poll: Some(Arc::clone(&resume_poll)),
+            },
+        )
+        .await;
+        let done = Arc::clone(
+            &sess
+                .active_turn
+                .lock()
+                .await
+                .as_ref()
+                .expect("old active turn")
+                .task
+                .as_ref()
+                .expect("old task")
+                .done,
+        );
+        let finished = done.notified();
+        tokio::pin!(finished);
+        finished.as_mut().enable();
+
+        // Poll the real guardian stop through detachment and cancellation, then suspend it
+        // at its graceful-stop wait. This makes the interleaving deterministic without
+        // racing a wall-clock sleep or adding a production test hook.
+        let stop = sess.abort_turn_if_active(&old_turn.sub_id, TurnAbortReason::Interrupted);
+        tokio::pin!(stop);
+        assert!(futures::poll!(&mut stop).is_pending());
+        assert!(sess.active_turn.lock().await.is_none());
+
+        sess.spawn_task(
+            Arc::clone(&new_turn),
+            Vec::new(),
+            StalePollTask {
+                polled: Arc::default(),
+                resume_poll: None,
+            },
+        )
+        .await;
+        let new_state = Arc::clone(
+            &sess
+                .active_turn
+                .lock()
+                .await
+                .as_ref()
+                .expect("replacement turn")
+                .turn_state,
+        );
+        let steer = TurnInput::UserInput {
+            content: vec![UserInput::Text {
+                text: "steering for the replacement".to_string(),
+                text_elements: Vec::new(),
+            }],
+            client_id: Some("replacement-steer".to_string()),
+        };
+        sess.input_queue
+            .extend_pending_input_for_turn_state(new_state.as_ref(), vec![steer.clone()])
+            .await;
+        let mut mail = InterAgentCommunication::new(
+            AgentPath::root(),
+            AgentPath::try_from("/root/worker").expect("agent path"),
+            Vec::new(),
+            "mail for the replacement".to_string(),
+            /*trigger_turn*/ true,
+        );
+        let message_id = mail.ensure_message_identity().to_string();
+        sess.input_queue
+            .enqueue_mailbox_communication(mail.clone(), Some("parent-turn".to_string()))
+            .await;
+
+        resume_poll.notify_one();
+        finished.await;
+        assert!(stop.await);
+        let stale = polled.lock().expect("polled lock").take();
+        let replacement = sess
+            .input_queue
+            .get_pending_input_for_running_task(&sess.active_turn, &new_turn.sub_id)
+            .await;
+        let taken_ids = new_state
+            .lock()
+            .await
+            .take_unrecorded_mailbox_message_ids(&[]);
+        sess.abort_all_tasks(TurnAbortReason::Interrupted).await;
+
+        assert_eq!(
+            stale,
+            Some(Vec::new()),
+            "the stopped turn stole replacement input"
+        );
+        assert_eq!(
+            replacement,
+            (
+                vec![steer, TurnInput::InterAgentCommunication(mail)],
+                Some("parent-turn".to_string()),
+            )
+        );
+        assert_eq!(taken_ids, vec![message_id]);
+    })
+    .await
+    .expect("bounded guardian/replacement interleaving");
+}

+# Issue #2046: guardian stop and replacement input
+
+## Scope and authority
+
+Bounded fix. [Product specification](../../docs/corbanu-product-spec.md), heading
+**Shipping MVP — LIVE**, Agent orchestration:
+“durable mailboxes, supervision, resume, and recovery.”
+
+Restore turn ownership of accepted steering and mailbox messages when guardian
+stops an old turn concurrently with admission of a new turn. No change to
+guardian decisions, authorization, mailbox persistence formats, or public APIs.
+No plan or sprint is required for this bounded fix.
+
+Base: `aeed8c39ab`. Test-only commit: `d503ffc7b7`.
+All implementation and validation take place in the supplied checkout on
+`work-b1`; no remote operations are authorized.
+
+## Regression contract
+
+`session::guardian_replacement_input_tests::guardian_stopping_turn_cannot_take_replacement_input`
+uses real Session task admission, targeted abort, and input-queue handling with
+a bounded synthetic task. It polls guardian's stop to its graceful cancellation
+wait, starts a replacement task, queues one user steer and one identified agent
+message, and lets the old task poll once before the stop finishes.
+
+Expected results:
+
+- The old task receives no replacement input.
+- The replacement receives both complete inputs, including the steer client ID,
+  message identity, and parent-turn provenance.
+- The replacement tracks the mailbox identity for later outcome handling.
+
+The whole case has a ten-second timeout and fixed-size inputs. No inference,
+third-party agent, or operator credential access is needed.
+
+## Qualification limits
+
+This is a local code/test result, not a qualified release or human-test handoff.
+True-TUI execution, independent code-blind packaged execution, live-repository
+qualification, and named-human acceptance are not performed in this local-only
+task. Those applicable product gates remain open before an unqualified functional
+handoff; automated tests do not substitute for them.
+
+## Implementation
+
+`core/src/session/input_queue.rs` requires the running caller's turn ID before
+taking any input. The ownership check, pending-input removal, mailbox drain, and
+mailbox identity tracking all retain the active-turn lock. The existing lock
+order is active turn, turn state, mailbox. Task admission retains its separate
+idle-slot behavior. `core/src/session/turn.rs` passes its own turn ID.
+
+The regression's assertions and scheduling are unchanged after the fix; only
+its polling calls and the shared synthetic task supply the now-required turn ID.
